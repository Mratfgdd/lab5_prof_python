class RecordValidationError(DataError):
    def __init__(
        self,
        message: str,
        *,
        line_number: int | None = None,
        field: str | None = None,
        code: str = "invalid",
    ) -> None:
        super().__init__(message)
        self.message = message
        self.line_number = line_number
        self.field = field
        self.code = code

    def __str__(self) -> str:
        where = f"рядок {self.line_number}" if self.line_number is not None else "запис"
        column = f", поле {self.field!r}" if self.field else ""
        return f"{where}{column}: {self.message} [{self.code}]"

def load_config(path: Path, environ: Mapping[str, str] | None = None) -> AppConfig:
    try:
        with path.open("r", encoding="utf-8") as file:
            raw = yaml.safe_load(file)
    except FileNotFoundError as error:
        raise ConfigurationError(f"Файл конфігурації не знайдено: {path}") from error
    except yaml.YAMLError as error:
        raise ConfigurationError(f"Некоректний YAML у {path.name}") from error
    except (OSError, UnicodeDecodeError) as error:
        raise ConfigurationError(f"Не вдалося прочитати {path}") from error

    if not isinstance(raw, dict):
        raise ConfigurationError("Корінь конфігурації має бути словником")
    apply_env_overrides(raw, os.environ if environ is None else environ)
    try:
        config = _build(raw, base=path.resolve().parent)
    except (KeyError, TypeError, ValueError, InvalidOperation, re.error) as error:
        raise ConfigurationError(f"Некоректна структура конфігурації: {error!r}") from error
    validate_config(config)
    return config

4. Logging

Лог пишеться у файл із ротацією (RotatingFileHandler) і в консоль. Повідомлення про некоректні записи містять лише номер рядка, поле, причину й код помилки, але не значення полів: імена студентів у log не потрапляють (це перевірено на справжньому файлі log під час генерації звіту й тестом). Приклад log – у розділі 7.

def configure_logging(config: LoggingConfig) -> None:
    config.path.parent.mkdir(parents=True, exist_ok=True)
    file_handler = RotatingFileHandler(
        config.path,
        maxBytes=config.max_bytes,
        backupCount=config.backup_count,
        encoding="utf-8",
    )
    console_handler = logging.StreamHandler(sys.stderr)
    formatter = logging.Formatter(FORMAT)
    for handler in (file_handler, console_handler):
        handler.setFormatter(formatter)
    logging.basicConfig(
        level=getattr(logging, config.level),
        handlers=[file_handler, console_handler],
        force=True,
    )

@contextmanager
def logged_operation(operation: str) -> Iterator[None]:
    logger.info("%s: початок", operation)
    start = time.perf_counter()
    try:
        yield
    except Exception:
        logger.error("%s: перервано через %.3f с", operation, time.perf_counter() - start)
        raise
    else:
        logger.info("%s: завершено за %.3f с", operation, time.perf_counter() - start)

5. Streaming reader, validation, strict / tolerant

Єдиний інтерфейс Importer (Protocol) має реалізації для CSV, JSON Lines (потокові), JSON і YAML (читаються цілком); формат обирається за розширенням. CSV читається рядок за рядком, а технічні помилки файлової системи, кодування та синтаксису перетворюються на DataImportError:

def read(self, path: Path) -> RowStream:
    try:
        with path.open("r", encoding="utf-8-sig", newline="") as file:
            reader = csv.DictReader(file)
            missing = set(REQUIRED_COLUMNS) - set(reader.fieldnames or ())
            if missing:
                raise DataImportError(f"{path.name}: бракує колонок {sorted(missing)}")
            for row in reader:
                yield reader.line_num, dict(row)
    except FileNotFoundError as error:
        raise DataImportError(f"Вхідний файл не знайдено: {path.name}") from error
    except UnicodeDecodeError as error:
        raise DataImportError(f"{path.name}: файл не в кодуванні UTF-8") from error
    except csv.Error as error:
        raise DataImportError(f"{path.name}: пошкоджений CSV") from error
    except OSError as error:
        raise DataImportError(f"Не вдалося прочитати {path.name}") from error

Валідатор перевіряє поля у фіксованому порядку, правила беруться з конфігурації:

def validate(self, line_number: int, row: RawRow) -> StudentRecord:
    student_id = self._student_id(row, line_number)
    first_name = self._name(row, "first_name", line_number)
    last_name = self._name(row, "last_name", line_number)
    group = self._text(row, "group", line_number)
    if not self._config.group_pattern.fullmatch(group):
        raise RecordValidationError(
            "код групи не відповідає формату",
            line_number=line_number,
            field="group",
            code="bad_group",
        )
    grade = self._grade(row, line_number)
    return StudentRecord(line_number, student_id, first_name, last_name, group, grade)

@staticmethod
def _student_id(row: RawRow, line_number: int) -> int:
    value = row.get("student_id")
    if value is None or isinstance(value, bool):
        raise RecordValidationError(
            "поле відсутнє", line_number=line_number, field="student_id", code="missing_field"
        )
    try:
        student_id = int(str(value).strip())
    except ValueError as error:
        raise RecordValidationError(
            "ідентифікатор не є цілим числом",
            line_number=line_number,
            field="student_id",
            code="bad_id",
        ) from error
    if student_id < 1:
        raise RecordValidationError(
            "ідентифікатор має бути додатним",
            line_number=line_number,
            field="student_id",
            code="bad_id",
        )
    return student_id

Поведінку з некоректним записом визначає об'єкт-політика: strict піднімає виняток (fail-fast), tolerant пише warning у log і рядок в errors.csv. Генератори import_records і filter_by_grade ведуть статистику й віддають записи по одному:

class StrictPolicy:
    def on_invalid(self, error: RecordValidationError, row: RawRow) -> None:
        raise error

class TolerantPolicy:
    def __init__(self, report: ErrorReport) -> None:
        self._report = report

    def on_invalid(self, error: RecordValidationError, row: RawRow) -> None:
        logger.warning("Некоректний запис пропущено: %s", error)
        self._report.add(error, row.get("student_id"))

def import_records(
    rows: Iterable[tuple[int, RawRow]],
    validator: RecordValidator,
    policy: ValidationPolicy,
    stats: ImportStatistics,
    progress_every: int = 10_000,
) -> Generator[StudentRecord, None, None]:
    for line_number, row in rows:
        stats.total += 1
        try:
            record = validator.validate(line_number, row)
        except RecordValidationError as error:
            stats.invalid += 1
            stats.errors[error.code] += 1
            policy.on_invalid(error, row)
        else:
            stats.valid += 1
            yield record
        if stats.total % progress_every == 0:
            logger.info("Прогрес: %d записів (некоректних %d)", stats.total, stats.invalid)

def filter_by_grade(
    records: Iterable[StudentRecord],
    minimum_grade: Decimal,
    stats: ImportStatistics,
    groups: Counter[str],
) -> Generator[StudentRecord, None, None]:
    for record in records:
        if record.average_grade < minimum_grade:
            stats.below_minimum += 1
            continue
        groups[record.group] += 1
        yield record

6. JSON serialization, atomic export, context managers

JsonExporter пише JSON-масив потоково, а Decimal серіалізується числом (custom encoder): оцінка має не більше двох знаків, тому float її не спотворює. AtomicWriter – власний class-based context manager: дані пишуться у тимчасовий файл у тому ж каталозі; лише після успішного блоку with виконуються fsync, резервна копія .bak і атомарний os.replace (з retry на випадок тимчасового блокування файла у Windows); при винятку тимчасовий файл видаляється. Generator-based logged_operation журналює тривалість операції.

class DecimalJSONEncoder(json.JSONEncoder):
    def default(self, o: object) -> object:
        if isinstance(o, Decimal):
            return float(o)
        return super().default(o)

class JsonExporter:
    def write(self, records: Iterable[StudentRecord], file: TextIO) -> int:
        count = 0
        file.write("[")
        for record in records:
            file.write(",\n  " if count else "\n  ")
            file.write(json.dumps(record.to_dict(), cls=DecimalJSONEncoder, ensure_ascii=False))
            count += 1
        file.write("\n]\n" if count else "]\n")
        return count
def __exit__(
    self,
    exc_type: type[BaseException] | None,
    exc: BaseException | None,
    traceback: TracebackType | None,
) -> None:
    assert self._file is not None and self._temp is not None
    try:
        try:
            if exc_type is None:
                self._file.flush()
                os.fsync(self._file.fileno())
        finally:
            self._file.close()
        if exc_type is not None:
            logger.warning("%s: запис скасовано, тимчасовий файл видалено", self.path.name)
            return
        if self.backup and self.path.exists():
            shutil.copy2(self.path, self.path.with_name(self.path.name + ".bak"))
        replace_file(self._temp, self.path)
    except OSError as error:
        raise DataExportError(f"Неможливо записати {self.path.name}") from error
    finally:
        self._temp.unlink(missing_ok=True)

У run_import_export писачі й генератори реєструються в ExitStack, причому генератори – через contextlib.closing: при зупинці strict-режиму посередині файла вхідний CSV закривається одразу, а не після збирання сміття (у Windows відкритий файл не можна видалити; це перевіряє тест test_strict_abort_releases_input_file).

Висновки

Створено надійний модуль імпорту/експорту студентів: дані перевіряються на межі системи, кожна очікувана помилка має свій тип і код, причина зберігається через chaining, ресурси звільняють context managers, а атомарний запис, як показав експеримент, не пошкоджує попередній output. Поведінка (strict/tolerant, мінімальний бал, формати) змінюється через YAML або змінні середовища без редагування коду, log не містить персональних даних, а streaming тримає пам'ять сталою незалежно від розміру файла.
